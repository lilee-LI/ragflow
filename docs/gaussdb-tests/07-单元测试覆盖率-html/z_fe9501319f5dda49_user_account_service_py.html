43def create_new_user(user_info: dict) -> dict: 

44 """ 

45 Add a new user, and create tenant, tenant llm, file folder for new user. 

46 :param user_info: { 

47 "email": <example@example.com>, 

48 "nickname": <str, "name">, 

49 "password": <decrypted password>, 

50 "login_channel": <enum, "password">, 

51 "is_superuser": <bool, role == "admin">, 

52 } 

53 :return: { 

54 "success": <bool>, 

55 "user_info": <dict>, # if true, return user_info 

56 } 

57 """ 

58 # generate user_id and access_token for user 

59 user_id = uuid.uuid1().hex 

60 user_info["id"] = user_id 

61 user_info["access_token"] = uuid.uuid1().hex 

62 # construct tenant info 

63 tenant = { 

64 "id": user_id, 

65 "name": user_info["nickname"] + "‘s Kingdom", 

66 "llm_id": settings.CHAT_MDL, 

67 "embd_id": settings.EMBEDDING_MDL, 

68 "asr_id": settings.ASR_MDL, 

69 "parser_ids": settings.PARSERS, 

70 "img2txt_id": settings.VISION_MDL, 

71 "rerank_id": settings.RERANK_MDL, 

72 } 

73 usr_tenant = { 

74 "tenant_id": user_id, 

75 "user_id": user_id, 

76 "invited_by": user_id, 

77 "role": UserTenantRole.OWNER, 

78 } 

79 # construct file folder info 

80 file_id = uuid.uuid1().hex 

81 file = { 

82 "id": file_id, 

83 "parent_id": file_id, 

84 "tenant_id": user_id, 

85 "created_by": user_id, 

86 "name": "/", 

87 "type": FileType.FOLDER.value, 

88 "size": 0, 

89 "location": "", 

90 } 

91 try: 

92 # tenant_llm = get_init_tenant_llm(user_id) 

93 

94 if not UserService.save(**user_info): 

95 return {"success": False} 

96 

97 TenantService.insert(**tenant) 

98 UserTenantService.insert(**usr_tenant) 

99 # TenantLLMService.insert_many(tenant_llm) 

100 FileService.insert(file) 

101 

102 return { 

103 "success": True, 

104 "user_info": user_info, 

105 } 

106 

107 except Exception as create_error: 

108 logging.exception(create_error) 

109 # rollback 

110 try: 

111 metadata_index_name = DocMetadataService._get_doc_meta_index_name(user_id) 

112 settings.docStoreConn.delete_idx(metadata_index_name, "") 

113 except Exception as e: 

114 logging.exception(e) 

115 try: 

116 TenantService.delete_by_id(user_id) 

117 except Exception as e: 

118 logging.exception(e) 

119 try: 

120 u = UserTenantService.query(tenant_id=user_id) 

121 if u: 

122 UserTenantService.delete_by_id(u[0].id) 

123 except Exception as e: 

124 logging.exception(e) 

125 try: 

126 FileService.delete_by_id(file["id"]) 

127 except Exception as e: 

128 logging.exception(e) 

129 # delete user row finally 

130 try: 

131 UserService.delete_by_id(user_id) 

132 except Exception as e: 

133 logging.exception(e) 

134 # reraise 

135 raise create_error

138def delete_user_data(user_id: str) -> dict: 

139 # use user_id to delete 

140 usr = UserService.filter_by_id(user_id) 

141 if not usr: 

142 return {"success": False, "message": f"{user_id} can't be found."} 

143 # check is inactive and not admin 

144 if usr.is_active == ActiveEnum.ACTIVE.value: 

145 return {"success": False, "message": f"{user_id} is active and can't be deleted."} 

146 if usr.is_superuser: 

147 return {"success": False, "message": "Can't delete the super user."} 

148 # tenant info 

149 tenants = UserTenantService.get_user_tenant_relation_by_user_id(usr.id) 

150 owned_tenant = [t for t in tenants if t["role"] == UserTenantRole.OWNER.value] 

151 

152 done_msg = "" 

153 try: 

154 # step1. delete owned tenant info 

155 if owned_tenant: 

156 done_msg += "Start to delete owned tenant.\n" 

157 tenant_id = owned_tenant[0]["tenant_id"] 

158 kb_ids = KnowledgebaseService.get_kb_ids(usr.id) 

159 # step1.1 delete dataset related file and info 

160 if kb_ids: 

161 # step1.1.1 delete files in storage, remove bucket 

162 for kb_id in kb_ids: 

163 if settings.STORAGE_IMPL.bucket_exists(kb_id): 

164 settings.STORAGE_IMPL.remove_bucket(kb_id) 

165 done_msg += f"- Removed {len(kb_ids)} dataset's buckets.\n" 

166 # step1.1.2 delete file and document info in db 

167 doc_ids = DocumentService.get_all_doc_ids_by_kb_ids(kb_ids) 

168 if doc_ids: 

169 for doc in doc_ids: 

170 try: 

171 DocMetadataService.delete_document_metadata(doc["id"], doc["kb_id"], tenant_id=None) 

172 except Exception as e: 

173 logging.warning(f"Failed to delete metadata for document {doc['id']}: {e}") 

174 

175 doc_delete_res = DocumentService.delete_by_ids([i["id"] for i in doc_ids]) 

176 done_msg += f"- Deleted {doc_delete_res} document records.\n" 

177 task_delete_res = TaskService.delete_by_doc_ids([i["id"] for i in doc_ids]) 

178 done_msg += f"- Deleted {task_delete_res} task records.\n" 

179 file_ids = FileService.get_all_file_ids_by_tenant_id(usr.id) 

180 if file_ids: 

181 file_delete_res = FileService.delete_by_ids([f["id"] for f in file_ids]) 

182 done_msg += f"- Deleted {file_delete_res} file records.\n" 

183 if doc_ids or file_ids: 

184 file2doc_delete_res = File2DocumentService.delete_by_document_ids_or_file_ids([i["id"] for i in doc_ids], [f["id"] for f in file_ids]) 

185 done_msg += f"- Deleted {file2doc_delete_res} document-file relation records.\n" 

186 # step1.1.3 delete chunk in es 

187 r = settings.docStoreConn.delete({"kb_id": kb_ids}, search.index_name(tenant_id), kb_ids) 

188 done_msg += f"- Deleted {r} chunk records.\n" 

189 kb_delete_res = KnowledgebaseService.delete_by_ids(kb_ids) 

190 done_msg += f"- Deleted {kb_delete_res} dataset records.\n" 

191 # step1.1.4 delete agents 

192 agent_delete_res = delete_user_agents(usr.id) 

193 done_msg += f"- Deleted {agent_delete_res['agents_deleted_count']} agent, {agent_delete_res['version_deleted_count']} versions records.\n" 

194 # step1.1.5 delete dialogs 

195 dialog_delete_res = delete_user_dialogs(usr.id) 

196 done_msg += f"- Deleted {dialog_delete_res['dialogs_deleted_count']} dialogs, {dialog_delete_res['conversations_deleted_count']} conversations, {dialog_delete_res['api_token_deleted_count']} api tokens, {dialog_delete_res['api4conversation_deleted_count']} api4conversations.\n" 

197 # step1.1.6 delete mcp server 

198 mcp_delete_res = MCPServerService.delete_by_tenant_id(usr.id) 

199 done_msg += f"- Deleted {mcp_delete_res} MCP server.\n" 

200 # step1.1.7 delete search 

201 search_delete_res = SearchService.delete_by_tenant_id(usr.id) 

202 done_msg += f"- Deleted {search_delete_res} search records.\n" 

203 # step1.2 delete tenant_langfuse 

204 # llm_delete_res = TenantLLMService.delete_by_tenant_id(tenant_id) 

205 # done_msg += f"- Deleted {llm_delete_res} tenant-LLM records.\n" 

206 langfuse_delete_res = TenantLangfuseService.delete_ty_tenant_id(tenant_id) 

207 done_msg += f"- Deleted {langfuse_delete_res} langfuse records.\n" 

208 try: 

209 metadata_index_name = DocMetadataService._get_doc_meta_index_name(tenant_id) 

210 settings.docStoreConn.delete_idx(metadata_index_name, "") 

211 done_msg += f"- Deleted metadata table {metadata_index_name}.\n" 

212 except Exception as e: 

213 logging.warning(f"Failed to delete metadata table for tenant {tenant_id}: {e}") 

214 done_msg += "- Warning: Failed to delete metadata table (continuing).\n" 

215 # step1.3 delete memory and messages 

216 user_memory = MemoryService.get_by_tenant_id(tenant_id) 

217 if user_memory: 

218 for memory in user_memory: 

219 if MessageService.has_index(tenant_id, memory.id): 

220 MessageService.delete_index(tenant_id, memory.id) 

221 done_msg += " Deleted memory index." 

222 memory_delete_res = MemoryService.delete_by_ids([m.id for m in user_memory]) 

223 done_msg += f"Deleted {memory_delete_res} memory datasets." 

224 # step1.4 delete own tenant 

225 tenant_delete_res = TenantService.delete_by_id(tenant_id) 

226 done_msg += f"- Deleted {tenant_delete_res} tenant.\n" 

227 # step2 delete user-tenant relation 

228 if tenants: 

229 # step2.1 delete docs and files in joined team 

230 joined_tenants = [t for t in tenants if t["role"] == UserTenantRole.NORMAL.value] 

231 if joined_tenants: 

232 done_msg += "Start to delete data in joined tenants.\n" 

233 created_documents = DocumentService.get_all_docs_by_creator_id(usr.id) 

234 if created_documents: 

235 # step2.1.1 delete files 

236 doc_file_info = File2DocumentService.get_by_document_ids([d["id"] for d in created_documents]) 

237 created_files = FileService.get_by_ids([f["file_id"] for f in doc_file_info]) 

238 if created_files: 

239 # step2.1.1.1 delete file in storage 

240 for f in created_files: 

241 settings.STORAGE_IMPL.rm(f.parent_id, f.location) 

242 done_msg += f"- Deleted {len(created_files)} uploaded file.\n" 

243 # step2.1.1.2 delete file record 

244 file_delete_res = FileService.delete_by_ids([f.id for f in created_files]) 

245 done_msg += f"- Deleted {file_delete_res} file records.\n" 

246 # step2.1.2 delete document-file relation record 

247 file2doc_delete_res = File2DocumentService.delete_by_document_ids_or_file_ids([d["id"] for d in created_documents], [f.id for f in created_files]) 

248 done_msg += f"- Deleted {file2doc_delete_res} document-file relation records.\n" 

249 # step2.1.3 delete chunks 

250 doc_groups = group_by(created_documents, "tenant_id") 

251 kb_grouped_doc = {k: group_by(v, "kb_id") for k, v in doc_groups.items()} 

252 # chunks in {'tenant_id': {'kb_id': [{'id': doc_id}]}} structure 

253 chunk_delete_res = 0 

254 kb_doc_info = {} 

255 for _tenant_id, kb_doc in kb_grouped_doc.items(): 

256 for _kb_id, docs in kb_doc.items(): 

257 chunk_delete_res += settings.docStoreConn.delete({"doc_id": [d["id"] for d in docs]}, search.index_name(_tenant_id), _kb_id) 

258 # record doc info 

259 if _kb_id in kb_doc_info.keys(): 

260 kb_doc_info[_kb_id]["doc_num"] += 1 

261 kb_doc_info[_kb_id]["token_num"] += sum([d["token_num"] for d in docs]) 

262 kb_doc_info[_kb_id]["chunk_num"] += sum([d["chunk_num"] for d in docs]) 

263 else: 

264 kb_doc_info[_kb_id] = {"doc_num": 1, "token_num": sum([d["token_num"] for d in docs]), "chunk_num": sum([d["chunk_num"] for d in docs])} 

265 done_msg += f"- Deleted {chunk_delete_res} chunks.\n" 

266 # step2.1.4 delete tasks 

267 task_delete_res = TaskService.delete_by_doc_ids([d["id"] for d in created_documents]) 

268 done_msg += f"- Deleted {task_delete_res} tasks.\n" 

269 # step2.1.5 delete document record 

270 doc_delete_res = DocumentService.delete_by_ids([d["id"] for d in created_documents]) 

271 done_msg += f"- Deleted {doc_delete_res} documents.\n" 

272 for doc in created_documents: 

273 try: 

274 DocMetadataService.delete_document_metadata(doc["id"], doc["kb_id"], doc["tenant_id"]) 

275 except Exception as e: 

276 logging.warning(f"Failed to delete metadata for document {doc['id']}: {e}") 

277 # step2.1.6 update dataset doc&chunk&token cnt 

278 for kb_id, doc_num in kb_doc_info.items(): 

279 KnowledgebaseService.decrease_document_num_in_delete(kb_id, doc_num) 

280 

281 # step2.2 delete relation 

282 user_tenant_delete_res = UserTenantService.delete_by_ids([t["id"] for t in tenants]) 

283 done_msg += f"- Deleted {user_tenant_delete_res} user-tenant records.\n" 

284 # step3 finally delete user 

285 user_delete_res = UserService.delete_by_id(usr.id) 

286 done_msg += f"- Deleted {user_delete_res} user.\nDelete done!" 

287 

288 return {"success": True, "message": f"Successfully deleted user. Details:\n{done_msg}"} 

289 

290 except Exception as e: 

291 logging.exception(e) 

292 return {"success": False, "message": "An internal error occurred during user deletion. Some operations may have completed.", "details": done_msg}

295def delete_user_agents(user_id: str) -> dict: 

296 """ 

297 use user_id to delete 

298 :return: { 

299 "agents_deleted_count": 1, 

300 "version_deleted_count": 2 

301 } 

302 """ 

303 agents_deleted_count, agents_version_deleted_count = 0, 0 

304 user_agents = UserCanvasService.get_all_agents_by_tenant_ids([user_id], user_id) 

305 if user_agents: 

306 agents_version = UserCanvasVersionService.get_all_canvas_version_by_canvas_ids([a["id"] for a in user_agents]) 

307 agents_version_deleted_count = UserCanvasVersionService.delete_by_ids([v["id"] for v in agents_version]) 

308 agents_deleted_count = UserCanvasService.delete_by_ids([a["id"] for a in user_agents]) 

309 return {"agents_deleted_count": agents_deleted_count, "version_deleted_count": agents_version_deleted_count}

312def delete_user_dialogs(user_id: str) -> dict: 

313 """ 

314 use user_id to delete 

315 :return: { 

316 "dialogs_deleted_count": 1, 

317 "conversations_deleted_count": 1, 

318 "api_token_deleted_count": 2, 

319 "api4conversation_deleted_count": 2 

320 } 

321 """ 

322 dialog_deleted_count, conversations_deleted_count, api_token_deleted_count, api4conversation_deleted_count = 0, 0, 0, 0 

323 user_dialogs = DialogService.get_all_dialogs_by_tenant_id(user_id) 

324 if user_dialogs: 

325 # delete conversation 

326 conversations = ConversationService.get_all_conversation_by_dialog_ids([ud["id"] for ud in user_dialogs]) 

327 conversations_deleted_count = ConversationService.delete_by_ids([c["id"] for c in conversations]) 

328 # delete api token 

329 api_token_deleted_count = APITokenService.delete_by_tenant_id(user_id) 

330 # delete api for conversation 

331 api4conversation_deleted_count = API4ConversationService.delete_by_dialog_ids([ud["id"] for ud in user_dialogs]) 

332 # delete dialog at last 

333 dialog_deleted_count = DialogService.delete_by_ids([ud["id"] for ud in user_dialogs]) 

334 return { 

335 "dialogs_deleted_count": dialog_deleted_count, 

336 "conversations_deleted_count": conversations_deleted_count, 

337 "api_token_deleted_count": api_token_deleted_count, 

338 "api4conversation_deleted_count": api4conversation_deleted_count, 

339 }
