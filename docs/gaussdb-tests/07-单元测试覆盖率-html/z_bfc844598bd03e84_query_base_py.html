20class QueryBase(ABC): 

21 @staticmethod 

22 def is_chinese(line): 

23 arr = re.split(r"[ \t]+", line) 

24 if len(arr) <= 3: 

25 return True 

26 e = 0 

27 for t in arr: 

28 if not re.match(r"[a-zA-Z]+$", t): 

29 e += 1 

30 return e * 1.0 / len(arr) >= 0.7 

31 

32 @staticmethod 

33 def sub_special_char(line): 

34 # Strip single quotes first to avoid Infinity's lexer treating them as string delimiters, 

35 # then escape remaining Infinity/Lucene special characters. 

36 return re.sub(r"([:\{\}/\[\]\-\*\?\"\(\)\|\+~\^])", r"\\\1", line.replace("'", "")).strip()

58 @staticmethod 

59 def add_space_between_eng_zh(txt): 

60 # (ENG/ENG+NUM) + ZH 

61 txt = re.sub(r"([A-Za-z]+[0-9]+)([\u4e00-\u9fa5]+)", r"\1 \2", txt) 

62 # ENG + ZH 

63 txt = re.sub(r"([A-Za-z])([\u4e00-\u9fa5]+)", r"\1 \2", txt) 

64 # ZH + (ENG/ENG+NUM) 

65 txt = re.sub(r"([\u4e00-\u9fa5]+)([A-Za-z]+[0-9]+)", r"\1 \2", txt) 

66 txt = re.sub(r"([\u4e00-\u9fa5]+)([A-Za-z])", r"\1 \2", txt) 

67 return txt
