608 def _drop_empty_metadata_table(cls, index_name: str, tenant_id: str) -> None: 

609 """ 

610 Check if metadata table is empty and drop it if so. 

611 Uses optimized count query instead of full search. 

612 This prevents accumulation of empty metadata tables. 

613 

614 Args: 

615 index_name: Metadata table/index name

624 return

626 logging.debug(f"[DROP EMPTY TABLE] Table {index_name} exists, checking if empty...") 

627

630 # The invocation lives inside the try/except so a future backend

644 select_fields=["id"], 

645 highlight_fields=[], 

646 condition={}, 

647 match_expressions=[], 

648 order_by=OrderByExpr(), 

649 offset=0,

651 index_names=index_name, 

652 knowledgebase_ids=[""], # Metadata tables don't filter by KB 

653 )

694 except Exception as e: 

695 # Log but don't fail - metadata deletion was successful 

696 logging.error(f"[DROP EMPTY TABLE] Failed to check/drop empty metadata table {index_name}: {e}")

1134 def _is_time_string(value: str) -> bool:

1136 if not isinstance(value, str):

1139 

1140 def _meta_value_type(value): 

1141 """Determine the type of a metadata value.""" 

1142 if value is None: 

1143 return None 

1144 if isinstance(value, list): 

1145 return "list" 

1146 if isinstance(value, bool): 

1147 return "string" 

1148 if isinstance(value, (int, float)): 

1149 return "number" 

1150 if isinstance(value, str) and _is_time_string(value): 

1151 return "time" 

1152 return "string" 

1153 

1154 try: 

1155 condition = {"kb_id": kb_id} 

1156 if doc_ids: 

1157 condition["id"] = doc_ids

1159 if not results: 

1160 return {} 

1161

1163 summary = {} 

1164 type_counter = {} 

1165

1167

1169 for doc_id, doc in cls._iter_search_results(results): 

1170 doc_meta = cls._extract_metadata(doc) 

1171

1175 if value_type:

1192 for k, v in summary.items(): 

1193 values = sorted([(val, cnt) for val, cnt in v.items()], key=lambda x: x[1], reverse=True) 

1194 type_counts = type_counter.get(k, {}) 

1195 value_type = "string" 

1196 if type_counts: 

1197 value_type = max(type_counts.items(), key=lambda item: item[1])[0] 

1198 result[k] = {"type": value_type, "values": values} 

1199 

1200 logging.debug(f"[METADATA SUMMARY] Final result: {result}") 

1201 return result 

1202 

1203 except Exception as e: 

1204 logging.error(f"Error getting metadata summary for KB {kb_id}: {e}") 

1205 return {}

1207 @classmethod 

1208 @DB.connection_context() 

1209 def batch_update_metadata(cls, kb_id: str, doc_ids: List[str], updates=None, deletes=None) -> int: 

1210 """ 

1211 Batch update metadata for documents in a knowledge base. 

1212 

1213 Args: 

1214 kb_id: Knowledge base ID 

1215 doc_ids: List of document IDs to update 

1216 updates: List of update operations, each with: 

1217 - key: field name to update 

1218 - value: new value 

1219 - match (optional): only update if current value matches this 

1220 deletes: List of delete operations, each with: 

1221 - key: field name to delete from 

1222 - value (optional): specific value to delete (if not provided, deletes the entire field) 

1223 

1224 Returns: 

1225 Number of documents updated 

1226 

1227 Examples: 

1228 updates = [{"key": "author", "value": "John"}] 

1229 updates = [{"key": "tags", "value": "new", "match": "old"}] # Replace "old" with "new" in tags list 

1230 deletes = [{"key": "author"}] # Delete entire author field 

1231 deletes = [{"key": "tags", "value": "obsolete"}] # Remove "obsolete" from tags list 

1232 """ 

1233 updates = updates or [] 

1234 deletes = deletes or [] 

1235 if not doc_ids: 

1236 return 0 

1237 

1238 def _normalize_meta(meta): 

1239 """Normalize metadata to a dict.""" 

1240 if isinstance(meta, str): 

1241 try: 

1242 meta = json.loads(meta) 

1243 except Exception: 

1244 return {} 

1245 if not isinstance(meta, dict): 

1246 return {} 

1247 return deepcopy(meta) 

1248 

1249 def _str_equal(a, b): 

1250 """Compare two values as strings.""" 

1251 return str(a) == str(b) 

1252 

1253 def _apply_updates(meta): 

1254 """Apply update operations to metadata.""" 

1255 changed = False 

1256 for upd in updates: 

1257 key = upd.get("key") 

1258 if not key: 

1259 continue 

1260 

1261 new_value = upd.get("value") 

1262 match_value = upd.get("match", None) 

1263 match_provided = match_value is not None and match_value != "" 

1264 

1265 if key not in meta: 

1266 if match_provided: 

1267 continue 

1268 meta[key] = dedupe_list(new_value) if isinstance(new_value, list) else new_value 

1269 changed = True 

1270 continue 

1271 

1272 if isinstance(meta[key], list): 

1273 if not match_provided: 

1274 # No match provided, append new_value to the list 

1275 if isinstance(new_value, list): 

1276 meta[key] = dedupe_list(meta[key] + new_value) 

1277 else: 

1278 meta[key] = dedupe_list(meta[key] + [new_value]) 

1279 changed = True 

1280 else: 

1281 # Replace items matching match_value with new_value 

1282 replaced = False 

1283 new_list = [] 

1284 for item in meta[key]: 

1285 if _str_equal(item, match_value): 

1286 new_list.append(new_value) 

1287 replaced = True 

1288 else: 

1289 new_list.append(item) 

1290 if replaced: 

1291 meta[key] = dedupe_list(new_list) 

1292 changed = True 

1293 else: 

1294 if not match_provided: 

1295 meta[key] = new_value 

1296 changed = True 

1297 else: 

1298 if _str_equal(meta[key], match_value): 

1299 meta[key] = new_value 

1300 changed = True 

1301 return changed 

1302 

1303 def _apply_deletes(meta): 

1304 """Apply delete operations to metadata.""" 

1305 changed = False 

1306 for d in deletes: 

1307 key = d.get("key") 

1308 if not key or key not in meta: 

1309 continue 

1310 value = d.get("value", None) 

1311 if isinstance(meta[key], list): 

1312 if value is None: 

1313 del meta[key] 

1314 changed = True 

1315 continue 

1316 new_list = [item for item in meta[key] if not _str_equal(item, value)] 

1317 if len(new_list) != len(meta[key]): 

1318 if new_list: 

1319 meta[key] = new_list 

1320 else: 

1321 del meta[key] 

1322 changed = True 

1323 else: 

1324 if value is None or _str_equal(meta[key], value): 

1325 del meta[key] 

1326 changed = True 

1327 return changed 

1328 

1329 try: 

1330 results = cls._search_metadata(kb_id, condition={"kb_id": kb_id, "id": doc_ids}) 

1331 if not results: 

1332 results = [] # Treat as empty list if None 

1333 

1334 updated_docs = 0 

1335 found_doc_ids = set() 

1336 

1337 logging.debug(f"[batch_update_metadata] Searching for doc_ids: {doc_ids}") 

1338 

1339 # Use helper to iterate over results 

1340 for doc_id, doc in cls._iter_search_results(results): 

1341 found_doc_ids.add(doc_id) 

1342 

1343 # Get current metadata 

1344 current_meta = cls._extract_metadata(doc) 

1345 meta = _normalize_meta(current_meta) 

1346 original_meta = deepcopy(meta) 

1347 

1348 logging.debug(f"[batch_update_metadata] Doc {doc_id}: current_meta={current_meta}, meta={meta}") 

1349 logging.debug(f"[batch_update_metadata] Updates to apply: {updates}, Deletes: {deletes}") 

1350 

1351 # Apply updates and deletes 

1352 changed = _apply_updates(meta) 

1353 logging.debug(f"[batch_update_metadata] After _apply_updates: changed={changed}, meta={meta}") 

1354 changed = _apply_deletes(meta) or changed 

1355 logging.debug(f"[batch_update_metadata] After _apply_deletes: changed={changed}, meta={meta}") 

1356 

1357 # Update if changed 

1358 if changed and meta != original_meta: 

1359 logging.debug(f"[batch_update_metadata] Updating doc_id: {doc_id}, meta: {meta}") 

1360 # If metadata is empty, delete the row entirely instead of keeping empty metadata 

1361 if not meta: 

1362 cls.delete_document_metadata(doc_id, kb_id, tenant_id=None) 

1363 else: 

1364 cls.update_document_metadata(doc_id, meta) 

1365 updated_docs += 1 

1366 

1367 # Handle documents that don't have metadata rows yet 

1368 # These documents weren't in the search results, so we need to insert new metadata for them 

1369 doc_ids_set = set(doc_ids) 

1370 missing_doc_ids = doc_ids_set - found_doc_ids 

1371 if missing_doc_ids and updates: 

1372 logging.debug(f"[batch_update_metadata] Inserting new metadata for documents without metadata rows: {missing_doc_ids}") 

1373 for doc_id in missing_doc_ids: 

1374 # Apply updates to create new metadata 

1375 meta = {} 

1376 _apply_updates(meta) 

1377 if meta: 

1378 # Only insert if there's actual metadata to add 

1379 cls.update_document_metadata(doc_id, meta) 

1380 updated_docs += 1 

1381 logging.debug(f"[batch_update_metadata] Inserted metadata for doc_id: {doc_id}, meta: {meta}") 

1382 

1383 logging.debug(f"[batch_update_metadata] KB: {kb_id}, doc_ids: {doc_ids}, updated: {updated_docs}") 

1384 return updated_docs 

1385 

1386 except Exception as e: 

1387 logging.error(f"Error in batch_update_metadata for KB {kb_id}: {e}") 

1388 return 0
