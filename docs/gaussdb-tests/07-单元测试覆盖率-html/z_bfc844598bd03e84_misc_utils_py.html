173def convert_bytes(size_in_bytes: int) -> str: 

174 """ 

175 Format size in bytes. 

176 """ 

177 if size_in_bytes == 0: 

178 return "0 B" 

179 

180 units = ["B", "KB", "MB", "GB", "TB", "PB"] 

181 i = 0 

182 size = float(size_in_bytes) 

183 

184 while size >= 1024 and i < len(units) - 1: 

185 size /= 1024 

186 i += 1 

187 

188 if i == 0 or size >= 100: 

189 return f"{size:.0f} {units[i]}" 

190 elif size >= 10: 

191 return f"{size:.1f} {units[i]}" 

192 else: 

193 return f"{size:.2f} {units[i]}"

196def once(func): 

197 """ 

198 A thread-safe decorator that ensures the decorated function runs exactly once, 

199 caching and returning its result for all subsequent calls. This prevents 

200 race conditions in multi-thread environments by using a lock to protect 

201 the execution state. 

202 

203 Args: 

204 func (callable): The function to be executed only once. 

205 

206 Returns: 

207 callable: A wrapper function that executes `func` on the first call 

208 and returns the cached result thereafter. 

209 

210 Example: 

211 @once 

212 def compute_expensive_value(): 

213 print("Computing...") 

214 return 42 

215 

216 # First call: executes and prints 

217 # Subsequent calls: return 42 without executing 

218 """ 

219 executed = False 

220 result = None 

221 lock = threading.Lock() 

222 

223 def wrapper(*args, **kwargs): 

224 nonlocal executed, result 

225 with lock: 

226 if not executed: 

227 executed = True 

228 result = func(*args, **kwargs) 

229 return result 

230 

231 return wrapper

244async def thread_pool_exec(func, *args, **kwargs): 

245 # loop.run_in_executor() submits the callable without propagating the caller's 

246 # contextvars (unlike asyncio.to_thread, which copies the context). Copy the 

247 # current context and run the callable inside it so ContextVars set by the 

248 # caller (e.g. tracing / per-request state) are visible in the worker thread. 

249 # 

250 # Use a short-lived executor per call instead of a shared singleton. Python 

251 # 3.13's executor reuse can deadlock in this environment when the same helper 

252 # is awaited repeatedly inside one event loop. 

253 loop = asyncio.get_running_loop() 

254 ctx = contextvars.copy_context() 

255 with ThreadPoolExecutor(max_workers=1) as executor: 

256 if kwargs: 256 ↛ 259line 256 didn't jump to line 259 because the condition on line 256 was always true

257 inner = functools.partial(func, *args, **kwargs) 

258 return await loop.run_in_executor(executor, ctx.run, inner) 

259 return await loop.run_in_executor(executor, ctx.run, func, *args)
