66 if not raw_api_key: 

67 return raw_api_key, None, None 

68 

69 try: 

70 parsed = json.loads(raw_api_key) 

71 except Exception: 

72 return raw_api_key, None, None 

73 

74 if not isinstance(parsed, dict): 

75 return raw_api_key, None, None

81 return parsed.get("api_key", raw_api_key), is_tools, raw_api_key

206def split_model_name(model_name: str): 

207 # Parse model_name: {model_name} or {model_name}@{factory_name} or {model_name}@{instance_name}@{factory_name} 

208 # 

209 # The composite key is right-anchored on the provider: the *last* '@'-separated 

210 # field is the factory/provider, the second-to-last is the instance (when 

211 # present), and everything to the left is the bare model name. Some model 

212 # names legitimately contain '@' characters themselves (e.g. LM Studio 

213 # embedding model IDs such as `text-embedding-nomic-embed-text-v1.5@q8_0`), 

214 # which produces composite keys like 

215 # `text-embedding-nomic-embed-text-v1.5@q8_0@lmstudio@LM-Studio`. 

216 # Use rsplit with maxsplit=2 from the right so any '@' characters embedded 

217 # in the leftmost model-name component are preserved as part of that name. 

218 parts = model_name.rsplit("@", 2) 

219 n = len(parts) 

220 if n == 3: 

221 pure_model_name, instance_name, provider_name = parts 

222 elif n == 2: 

223 pure_model_name, provider_name = parts 

224 instance_name = "default" 

225 else: 

226 pure_model_name = parts[0] 

227 provider_name = "" 

228 instance_name = "" 

229 return pure_model_name, instance_name, provider_name

262def get_model_config_from_provider_instance(tenant_id, model_type: str | enum.Enum, model_name: str): 

263 pure_model_name, instance_name, provider_name = split_model_name(model_name) 

264 model_type_val = model_type if isinstance(model_type, str) else model_type.value 

265 # Builtin embedding model 

266 compose_profiles = os.getenv("COMPOSE_PROFILES", "") 

267 is_tei_builtin_embedding = ( 

268 model_type_val == LLMType.EMBEDDING.value and "tei-" in compose_profiles and pure_model_name == os.getenv("TEI_MODEL", "") and (provider_name == "Builtin" or not provider_name) 

269 ) 

270 if is_tei_builtin_embedding: 

271 # configured local embedding model 

272 embedding_cfg = settings.EMBEDDING_CFG 

273 return { 

274 "llm_factory": "Builtin", 

275 "api_key": embedding_cfg["api_key"], 

276 "llm_name": pure_model_name, 

277 "api_base": embedding_cfg["base_url"], 

278 "model_type": LLMType.EMBEDDING.value, 

279 } 

280 

281 provider_obj = TenantModelProviderService.get_by_tenant_id_and_provider_name(tenant_id, provider_name) 

282 if not provider_obj: 

283 raise LookupError(f"Provider {provider_name} not found for model {model_name}.") 

284 instance_obj = _resolve_instance_for_model(provider_obj, instance_name, model_name) 

285 model_obj = TenantModelService.get_by_provider_id_and_instance_id_and_model_type_and_model_name(provider_obj.id, instance_obj.id, model_type_val, pure_model_name) 

286 

287 api_key, is_tool, api_key_payload = _decode_api_key_config(instance_obj.api_key) 

288 extra_fields = json.loads(instance_obj.extra) if instance_obj.extra else {} 

289 

290 if model_obj: 

291 if model_obj.status == ActiveStatusEnum.INACTIVE.value: 

292 raise LookupError(f"Model {model_name} is disabled.") 

293 if model_obj.status == ActiveStatusEnum.UNSUPPORTED.value: 

294 raise LookupError(f"Model {model_name} cannot be used as {model_type_val} model.") 

295 

296 model_extra = json.loads(model_obj.extra) if model_obj.extra else {} 

297 llm_info = _lookup_factory_llm_info(provider_obj.provider_name, pure_model_name, extra_fields) 

298 if "max_tokens" in model_extra: 

299 max_tokens = model_extra["max_tokens"] 

300 else: 

301 max_tokens = (llm_info or {}).get("max_tokens", 8192) 

302 model_config = { 

303 "llm_factory": provider_obj.provider_name, 

304 "api_key": api_key, 

305 "llm_name": model_obj.model_name, 

306 "api_base": extra_fields.get("base_url", ""), 

307 "model_type": model_type_val, 

308 "is_tools": model_extra.get("is_tools", is_tool), 

309 "max_tokens": max_tokens, 

310 } 

311 if provider_name.lower() == "somark": 

312 # SoMark/OCR factories read parser config (somark_*, parse_method, ...) 

313 # from model_config["extra"]; see tenant_llm_service.LLMBundle OCR path. 

314 model_config["extra"] = model_extra 

315 

316 if api_key_payload is not None: 

317 model_config["api_key_payload"] = api_key_payload 

318 

319 return model_config 

320 else: 

321 raise LookupError(f"Model {model_name} not found for model {model_type_val}")

324def get_model_config_by_id(tenant_id: str, model_type: str | enum.Enum, model_id: str): 

325 """Get model config from tenant_model by its id (CharField PK).""" 

326 model_type_val = model_type if isinstance(model_type, str) else model_type.value 

327 model_type_bin = calculate_model_type(model_type_val) 

328 exist, model_obj = TenantModelService.get_by_id(model_id) 

329 if not exist: 

330 raise LookupError(f"TenantModel id={model_id} not found.") 

331 if model_obj.status == ActiveStatusEnum.INACTIVE.value: 

332 raise LookupError(f"TenantModel id={model_id} is disabled.") 

333 if model_obj.status == ActiveStatusEnum.UNSUPPORTED.value: 

334 raise LookupError(f"TenantModel id={model_id} cannot be used as {model_type_val} model.") 

335 if not (model_obj.model_type & model_type_bin): 

336 raise LookupError(f"TenantModel id={model_id} cannot be used as {model_type_val} model.") 

337 

338 ok, provider_obj = TenantModelProviderService.get_by_id(model_obj.provider_id) 

339 if not ok: 

340 raise LookupError(f"Provider id={model_obj.provider_id} not found for model id={model_id}.") 

341 

342 # Validate that tenant_id owns the provider or is a joined tenant of the provider's owner. 

343 if tenant_id != provider_obj.tenant_id: 

344 joined_tenants = TenantService.get_joined_tenants_by_user_id(tenant_id) 

345 joined_tenant_ids = [t["tenant_id"] for t in joined_tenants] 

346 if provider_obj.tenant_id not in joined_tenant_ids: 

347 raise LookupError(f"Tenant {tenant_id} has no access to provider owned by tenant {provider_obj.tenant_id}.") 

348 

349 ok, instance_obj = TenantModelInstanceService.get_by_id(model_obj.instance_id) 

350 if not ok: 

351 raise LookupError(f"Instance id={model_obj.instance_id} not found for model id={model_id}.") 

352 

353 api_key, is_tool, api_key_payload = _decode_api_key_config(instance_obj.api_key) 

354 extra_fields = json.loads(instance_obj.extra) if instance_obj.extra else {} 

355 model_extra = json.loads(model_obj.extra) if model_obj.extra else {} 

356 

357 model_config = { 

358 "llm_factory": provider_obj.provider_name, 

359 "api_key": api_key, 

360 "llm_name": model_obj.model_name, 

361 "api_base": extra_fields.get("base_url", ""), 

362 "model_type": model_type_val, 

363 "is_tools": model_extra.get("is_tools", is_tool), 

364 "max_tokens": model_extra.get("max_tokens") or 8192, 

365 } 

366 if provider_obj.provider_name.lower() == "somark": 

367 model_config["extra"] = model_extra 

368 

369 if api_key_payload is not None: 

370 model_config["api_key_payload"] = api_key_payload 

371 

372 return model_config

375def resolve_model_id(tenant_id: str, model_type: str | enum.Enum, model_name: str) -> str | None: 

376 """Given a tenant_id, model_type and model_name (e.g. 'model@instance@provider'), 

377 look up the corresponding tenant_model.id. Returns None if not found.""" 

378 pure_model_name, instance_name, provider_name = split_model_name(model_name) 

379 model_type_val = model_type if isinstance(model_type, str) else model_type.value 

380 

381 # Builtin TEI embedding — no tenant_model row exists 

382 compose_profiles = os.getenv("COMPOSE_PROFILES", "") 

383 is_tei_builtin_embedding = ( 

384 model_type_val == LLMType.EMBEDDING.value and "tei-" in compose_profiles and pure_model_name == os.getenv("TEI_MODEL", "") and (provider_name == "Builtin" or not provider_name) 

385 ) 

386 if is_tei_builtin_embedding: 

387 return None 

388 

389 if not provider_name: 

390 raise LookupError(f"Provider name is required to resolve model id for {model_name}.") 

391 

392 provider_obj = TenantModelProviderService.get_by_tenant_id_and_provider_name(tenant_id, provider_name) 

393 if not provider_obj: 

394 raise LookupError(f"Provider {provider_name} not found for model {model_name}.") 

395 

396 instance_obj = _resolve_instance_for_model(provider_obj, instance_name, model_name) 

397 model_obj = TenantModelService.get_by_provider_id_and_instance_id_and_model_type_and_model_name(provider_obj.id, instance_obj.id, model_type_val, pure_model_name) 

398 if not model_obj: 

399 raise LookupError(f"Model {model_name} not found for type {model_type_val}.") 

400 return model_obj.id

414def ensure_tenant_model_ids_for_params(tenant_id: str, params: dict) -> dict: 

415 """For each model-name field present in *params*, resolve the corresponding 

416 tenant_model id if the id field is not already present. 

417 

418 Modifies *params* in-place (adds ``tenant_*_id`` keys) and returns it. 

419 Silently skips resolution when the model is not found in tenant_model 

420 (e.g. builtin TEI embedding). 

421 

422 Typical usage at API entry points: 

423 

424 req = await get_request_json() 

425 ensure_tenant_model_ids_for_params(current_user.id, req) 

426 # req now has tenant_llm_id / tenant_embd_id etc. filled in 

427 """ 

428 for name_field, (model_type, id_field) in _MODEL_NAME_TO_ID_FIELD_MAP.items(): 

429 if name_field in params and id_field not in params: 

430 try: 

431 params[id_field] = resolve_model_id(tenant_id, model_type, params[name_field]) 

432 except LookupError: 

433 logger.debug("Could not resolve %s → %s for tenant %s, skipping", name_field, id_field, tenant_id) 

434 return params
